# Official MCP Python SDK (pulls in starlette + uvicorn used by server.py).
# Capped below 2.0 on purpose: mcp 2.x removed `mcp.server.fastmcp` (the FastMCP
# entry point this server is built on) in favour of `mcp.server.mcpserver`, so an
# unpinned bound would break the image on the next no-cache rebuild. Migrating to
# 2.x is a separate change; Dependabot will still raise it as a major bump.
mcp>=1.28.1,<2
# CalDAV client — WebDAV + iCalendar over HTTP; talks to iCloud and other servers.
caldav>=3.2.1
# iCalendar object model used to build and parse VEVENTs.
icalendar>=7.2.2
# JWT verification of Pomerium's identity assertion (ES256 needs cryptography).
pyjwt[crypto]>=2.13.0
# Fetching subscribed ICS feeds (Apple "subscribed calendars" are not reachable
# over CalDAV, so they are pulled straight from their HTTP(S) URL).
httpx>=0.27.0
# Expanding RRULEs in those feeds so recurring events land on every occurrence.
recurring-ical-events>=3.2.0
# Locking the persisted subscription pull list against concurrent tool calls.
filelock>=3.13.0
