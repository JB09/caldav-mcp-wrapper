# Official MCP Python SDK (pulls in starlette + uvicorn used by server.py).
# Capped below 3.0: server.py is built on `mcp.server.mcpserver.MCPServer`, which
# replaced the 1.x `mcp.server.fastmcp.FastMCP` entry point in 2.0. Majors are
# excluded from Dependabot automerge, so a 3.x bump lands as a reviewable PR.
mcp>=2.0.0,<3
# CalDAV client — WebDAV + iCalendar over HTTP; talks to iCloud and other servers.
caldav>=3.2.1
# iCalendar object model used to build and parse VEVENTs.
icalendar>=7.3.0
# JWT verification of Pomerium's identity assertion (ES256 needs cryptography).
pyjwt[crypto]>=2.13.0
# Fetching subscribed ICS feeds (Apple "subscribed calendars" are not reachable
# over CalDAV, so they are pulled straight from their HTTP(S) URL). This is the
# same client mcp 2.x uses, so the image ships one HTTP stack rather than two —
# but it is declared here in its own right, not leaned on transitively, since the
# SDK is free to change what it depends on. Note httpx2 verifies TLS against the
# *system* trust store (via truststore) rather than a bundled certifi CA set, so
# the base image must provide ca-certificates — the smoke test asserts it does.
httpx2>=2.12.0
# Expanding RRULEs in those feeds so recurring events land on every occurrence.
recurring-ical-events>=3.8.2
# Locking the persisted subscription pull list against concurrent tool calls.
filelock>=3.32.3
